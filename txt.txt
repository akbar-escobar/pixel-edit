        let fillGap: { x: number, y: number }
        const pos = this.state.ctxDrawXY(x, y)

        if (this.prev.x !== -1 && this.prev.y !== -1) {
            if (Math.abs(pos.x - this.prev.x) >= 2) {
                fillGap = {
                    x: this.prev.x + (pos.x - this.prev.x) - 1,
                    y: pos.y
                }
                this.isFill = true
            } else if (Math.abs(pos.y - this.prev.y) >= 2) {
                fillGap = {
                    x: pos.x,
                    y: this.prev.y + (pos.y - this.prev.y) - 1
                }
                this.isFill = true
            } else if (
                Math.abs(pos.x - this.prev.x) >= 2 &&
                Math.abs(pos.y - this.prev.y) >= 2
            ) {
                fillGap = {
                    x: this.prev.x + (pos.x - this.prev.x) - 1,
                    y: this.prev.y + (pos.y - this.prev.y) - 1
                }
                this.isFill = true
            } else {
                fillGap = { x: pos.x, y: pos.y }
                this.isFill = false
            }
        } else fillGap = { x: pos.x, y: pos.y }
        this.prev = { x: pos.x, y: pos.y }

        if (this.state.toolCond === "brush") {
            if (this.isFill) this.brush.draw(fillGap.x, fillGap.y, "orange")
            else this.brush.draw(fillGap.x, fillGap.y, this.state.brushCol)
        }

        if (this.state.toolCond === "eraser") {
            this.eraser.erase(pos.x, pos.y)
        }
    }
